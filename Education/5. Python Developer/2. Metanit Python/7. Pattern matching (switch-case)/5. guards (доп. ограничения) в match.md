## Guards (ограничения шаблонов)

Guards (или ограничения) позволяют добавить **дополнительные условия** к шаблону в `match`. Они указываются после шаблона с ключевым словом `if`.

---

### Синтаксис
```python
case шаблон if условие:
    действие

- Шаблон должен совпасть **И** условие должно быть `True`
    
- Если условие `False` — шаблон не подходит
```

---
### Базовый пример: проверка возраста

```python
class Person:
    def __init__(self, name, age):
        self.name = name
        self.age = age
        
def enter(person):
    match person:
        case Person(name=name, age=age) if age < 18:
            print(f"{name}, доступ запрещен")
        case Person(name=name):
            print(f"{name}, добро пожаловать!")
            
enter(Person("Tom", 37))   # Tom, добро пожаловать!
enter(Person("Sam", 12))   # Sam, доступ запрещен
```

### Несколько условий

```python
def enter(person):
    match person:
        case Person(name=name, age=age) if age < 18:
            print(f"{name}, доступ запрещен")
        case Person(name=name, age=age) if age < 22:
            print(f"{name}, доступ ограничен")
        case Person(name=name):
            print(f"{name}, у вас полноценный доступ!")
            
enter(Person("Tom", 37))   # Tom, у вас полноценный доступ!
enter(Person("Bob", 20))   # Bob, доступ ограничен
enter(Person("Sam", 12))   # Sam, доступ запрещен
```

### Сложные условия

```python
def check_data(data):
    match data:
        case name, age if name == "admin" or age not in range(1, 101):
            print("Некорректные значения")
        case name, age:
            print(f"Данные проверены. Name: {name}  Age: {age}")
            
check_data(("admin", -45))   # Некорректные значения
check_data(("Tom", 37))      # Данные проверены. Name: Tom  Age: 37
```

---
### Guard с несколькими переменными

```python
def compare_points(point1, point2):
    match (point1, point2):
        case ((x1, y1), (x2, y2)) if x1 == x2 and y1 == y2:
            print("Точки совпадают")
        case ((x1, y1), (x2, y2)) if x1 == x2:
            print("Точки на одной вертикали")
        case ((x1, y1), (x2, y2)) if y1 == y2:
            print("Точки на одной горизонтали")
        case _:
            print("Точки разные")
            
compare_points((0, 0), (0, 0))     # Точки совпадают
compare_points((0, 5), (0, 10))    # Точки на одной вертикали
compare_points((5, 0), (10, 0))    # Точки на одной горизонтали
```

---
### Guard со списками

```python
def process_numbers(nums):
    match nums:
        case [x, y] if x + y > 10:
            print(f"Сумма {x} + {y} > 10")
        case [x, y]:
            print(f"Сумма {x} + {y} <= 10")
        case _:
            print("Не список из двух элементов")
            
process_numbers([8, 5])   # Сумма 8 + 5 > 10
process_numbers([3, 4])   # Сумма 3 + 4 <= 10
```

### Guard со словарями

```python
def validate_user(data):
    match data:
        case {"name": name, "age": age} if age >= 18 and len(name) > 2:
            print(f"{name} прошел проверку")
        case {"name": name, "age": age} if age < 18:
            print(f"{name}: слишком молод")
        case {"name": name} if len(name) <= 2:
            print(f"Имя '{name}' слишком короткое")
        case _:
            print("Неверный формат данных")
            
validate_user({"name": "Tom", "age": 25})      # Tom прошел проверку
validate_user({"name": "Sam", "age": 12})      # Sam: слишком молод
validate_user({"name": "T", "age": 25})        # Имя 'T' слишком короткое
```

### Guard с извлечением данных

```python
class Order:
    def __init__(self, amount, status, customer):
        self.amount = amount
        self.status = status
        self.customer = customer
        
def process_order(order):
    match order:
        case Order(amount=amount, status="pending", customer=customer) if amount > 1000:
            print(f"Крупный заказ от {customer} на сумму {amount} — нужна проверка")
        case Order(amount=amount, status="pending", customer=customer) if amount <= 1000:
            print(f"Заказ от {customer} на сумму {amount} — автоматическое одобрение")
        case Order(amount=amount, status="approved", customer=customer):
            print(f"Заказ от {customer} на сумму {amount} уже одобрен")
        case _:
            print("Неизвестный заказ")
            
order1 = Order(1500, "pending", "Tom")
order2 = Order(500, "pending", "Bob")
order3 = Order(2000, "approved", "Sam")

process_order(order1)  
# Крупный заказ от Tom на сумму 1500 — нужна проверка

process_order(order2)  
# Заказ от Bob на сумму 500 — автоматическое одобрение

process_order(order3)  
# Заказ от Sam на сумму 2000 уже одобрен
```

## Итог

| Конструкция                       | Описание                                 |
| --------------------------------- | ---------------------------------------- |
| `case pattern if condition:`      | Шаблон + дополнительное условие          |
| `if age < 18`                     | Простое условие                          |
| `if x == y and z > 0`             | Составное условие                        |
| `if key in dict or value is None` | Сложное условие с несколькими проверками |
