Selenium WebDriver предоставляет **возможность выполнять произвольный JavaScript-код в контексте текущей страницы** через интерфейс **`JavascriptExecutor`**. 

Это мощный инструмент, который позволяет делать то, что невозможно или сложно сделать через стандартные методы WebDriver.

**Через него вы можете:
- Выполнять любой JavaScript-код на странице
    
- Получать данные, недоступные через стандартные методы
    
- Взаимодействовать с элементами, которые не поддаются обычным методам
    
- Управлять прокруткой, анимациями и DOM

---
## Структура JavascriptExecutor

```java
// Получение экземпляра JavascriptExecutor
JavascriptExecutor js = (JavascriptExecutor) driver;
// Основные методы
js.executeScript(script, args...)        // Выполнение синхронного JS
js.executeAsyncScript(script, args...)   // Выполнение асинхронного JS
```

## Получение JavascriptExecutor

### Способы получения

```java
import org.openqa.selenium.JavascriptExecutor;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.chrome.ChromeDriver;

public class GetJavascriptExecutor {
    public static void main(String[] args) {
        WebDriver driver = new ChromeDriver();
        
        // Способ 1: Прямое приведение типа (самый распространенный)
        JavascriptExecutor js = (JavascriptExecutor) driver;
        
        // Способ 2: В одну строку
        ((JavascriptExecutor)driver)
	        .executeScript("alert('Привет!')");
        
        // Способ 3: С проверкой (если драйвер не поддерживает JS)
        if (driver instanceof JavascriptExecutor) {
            JavascriptExecutor jsExecutor = (JavascriptExecutor) driver;
            jsExecutor.executeScript("console.log('JS поддерживается')");
        }
        
        driver.quit();
    }
}
```

## Базовые операции с JavaScript

### Базовый синтаксис

```java
public class BasicJavascriptExample {

    public static void main(String[] args) {
    
        WebDriver driver = new ChromeDriver();
        JavascriptExecutor js = (JavascriptExecutor) driver;
        
        driver.get("https://example.com");
        // 1. Выполнение простого скрипта (без возвращаемого значения)
        js.executeScript("console.log('Hello from Selenium!')");
        
        // 2. Выполнение скрипта с возвращаемым значением
        String title = (String) js.executeScript(
	        "return document.title"
	    );
        System.out.println("Заголовок: " + title);
        
        // 3. Выполнение скрипта с параметрами
        String elementId = "someId";
        js.executeScript(
	        "document.getElementById(arguments[0]).style.border = '2px solid red'", 
	        elementId
	    );
	    
        // 4. Возврат сложных объектов (Map)
        Map<String, Object> pageInfo = (Map<String, Object>) js.executeScript(
            "return { " +
            "  title: document.title, " +
            "  url: document.URL, " +
            "  width: window.innerWidth " +
            "}"
        );
        System.out.println("Title: " + pageInfo.get("title"));
        System.out.println("URL: " + pageInfo.get("url"));
        System.out.println("Width: " + pageInfo.get("width"));
        driver.quit();
    }
}
```

## Работа с DOM и элементами

### Поиск и взаимодействие с элементами через JS

```java
public class DomJavascriptExample {
    public static void main(String[] args) {
        WebDriver driver = new ChromeDriver();
        
        JavascriptExecutor js = (JavascriptExecutor) driver;
        driver.get("https://example.com");
        
        // 1. Поиск элемента через JS
        WebElement element = (WebElement) js.executeScript(
            "return document.getElementById('someId')"
        );
        // 2. Поиск нескольких элементов
        List<WebElement> elements = (List<WebElement>) js.executeScript(
            "return document.querySelectorAll('.someClass')"
        );
        // 3. Изменение CSS элемента
        js.executeScript(
            "arguments[0].style.backgroundColor = 'yellow'; " +
            "arguments[0].style.border = '3px solid red'",
            element
        );
        // 4. Изменение атрибута
        js.executeScript(
            "arguments[0].setAttribute('disabled', 'disabled')",
            element
        );
        // 5. Удаление атрибута
        js.executeScript(
            "arguments[0].removeAttribute('readonly')",
            element
        );
        // 6. Добавление класса
        js.executeScript(
            "arguments[0].classList.add('highlighted')",
            element
        );
        // 7. Удаление класса
        js.executeScript(
            "arguments[0].classList.remove('hidden')",
            element
        );
        driver.quit();
    }
}
```

## Прокрутка страницы (Scrolling)

### Виды прокрутки

```java
public class ScrollExample {
    public static void main(String[] args) {
        WebDriver driver = new ChromeDriver();
        
        JavascriptExecutor js = (JavascriptExecutor) driver;
        driver.get("https://example.com");
        
        // 1. Прокрутка вниз на 500 пикселей
        js.executeScript("window.scrollBy(0, 500)");
        
        // 2. Прокрутка вверх на 200 пикселей
        js.executeScript("window.scrollBy(0, -200)");
        
        // 3. Прокрутка в самый низ страницы
        js.executeScript(
	        "window.scrollTo(0, document.body.scrollHeight)");
        
        // 4. Прокрутка в самый верх страницы
        js.executeScript("window.scrollTo(0, 0)");
        
        // 5. Прокрутка к элементу (с выравниванием вверху)
        WebElement element = driver.findElement(
	        By.id("targetElement")
	    );
	    
        js.executeScript(
	        "arguments[0].scrollIntoView(true)",
	        element);
	        
        // 6. Прокрутка к элементу (с выравниванием внизу)
        js.executeScript(
	        "arguments[0].scrollIntoView(false)",
	        element);
	        
        // 7. Прокрутка к элементу с плавной анимацией
        js.executeScript(
            "arguments[0].scrollIntoView({behavior: 'smooth', block: 'center'})",
            element
        );
        
        // 8. Прокрутка на определенное расстояние от элемента
        js.executeScript(
            "arguments[0].scrollIntoView(); " +
            "window.scrollBy(0, -100)",
            element
        );
        // 9. Получение текущей позиции прокрутки
        Long scrollY = (Long) js.executeScript(
	        "return window.pageYOffset");
        System.out.println("Текущая прокрутка: " + scrollY);
        
        // 10. Проверка, виден ли элемент
        Boolean isVisible = (Boolean) js.executeScript(
            "var rect = arguments[0].getBoundingClientRect(); " +
            "return rect.top >= 0 && rect.bottom <= window.innerHeight",
            element
        );
        System.out.println("Элемент виден: " + isVisible);
        driver.quit();
    }
}
```

## Получение информации со страницы

### Извлечение данных через JS

```java
public class GetInfoExample {
    public static void main(String[] args) {
        WebDriver driver = new ChromeDriver();
        
        JavascriptExecutor js = (JavascriptExecutor) driver;
        
        driver.get("https://example.com");
        
        // 1. Получение заголовка
        String title = (String) js.executeScript("return document.title");
        
        // 2. Получение URL
        String url = (String) js.executeScript("return document.URL");
        
        // 3. Получение размера окна
        Long width = (Long) js.executeScript(
	        "return window.innerWidth");
	        
        Long height = (Long) js.executeScript(
	        "return window.innerHeight");
	        
        System.out.println("Размер окна: " + width + "x" + height);
        
        // 4. Получение всех ссылок на странице
        List<String> links = (List<String>) js.executeScript(
            "var links = []; " +
            "document.querySelectorAll('a').forEach(function(el) { " +
            "  links.push(el.href); " +
            "}); " +
            "return links;"
        );
        
        // 5. Получение всех изображений
        List<String> images = (List<String>) js.executeScript(
            "var imgs = []; " +
            "document.querySelectorAll('img').forEach(function(el) { " +
            "  imgs.push(el.src); " +
            "}); " +
            "return imgs;"
        );
        
        
        // 6. Получение текста всех параграфов
        List<String> paragraphs = (List<String>) js.executeScript(
            "var texts = []; " +
            "document.querySelectorAll('p').forEach(function(el) { " +
            "  texts.push(el.textContent); " +
            "}); " +
            "return texts;"
        );
        
        
        // 7. Получение атрибутов элемента
        WebElement element = driver.findElement(By.id("someId"));
        String attrValue = (String) js.executeScript(
            "return arguments[0].getAttribute('data-custom')",
            element
        );
        
        
        // 8. Получение вычисленных стилей элемента
        String color = (String) js.executeScript(
            "return window.getComputedStyle(arguments[0]).color",
            element
        );
        
        
        // 9. Получение всех cookies
        String cookies = (String) js.executeScript(
	        "return document.cookie");
        
        
        // 10. Получение информации о производительности
        Long loadTime = (Long) js.executeScript(
            "return performance.timing.loadEventEnd - performance.timing.navigationStart"
        );
        System.out.println(
	        "Время загрузки страницы: " + loadTime + "ms");
        driver.quit();
    }
}
```

## Клики и взаимодействие через JS

### Когда стандартный клик не работает

```java
public class ClickJavascriptExample {
    public static void main(String[] args) {
        WebDriver driver = new ChromeDriver();
        
        JavascriptExecutor js = (JavascriptExecutor) driver;
        
        driver.get("https://example.com");
        
        WebElement element = 
	        driver.findElement(By.id("buttonId"));
	        
        // 1. Клик через JavaScript (обходит блокировки)
        js.executeScript("arguments[0].click()", element);
        
        // 2. Клик через dispatchEvent (имитация реального клика)
        js.executeScript(
            "var event = new MouseEvent('click', { " +
            "  view: window, " +
            "  bubbles: true, " +
            "  cancelable: true " +
            "}); " +
            "arguments[0].dispatchEvent(event)",
            element
        );
        
        // 3. Клик по элементу, который перекрыт другим (сделать видимым)
        js.executeScript(
            "arguments[0].style.display = 'block'; " +
            "arguments[0].style.visibility = 'visible'; " +
            "arguments[0].click()",
            element
        );
        
        // 4. Клик по элементу в iframe (сначала переключиться в iframe)
        driver.switchTo().frame("frameId");
        js.executeScript("arguments[0].click()", element);
        driver.switchTo().defaultContent();
        driver.quit();
    }
}
```

## Асинхронное выполнение JavaScript

### executeAsyncScript()

Асинхронный скрипт выполняется не мгновенно, а может иметь задержку. Он должен вызвать callback-функцию по завершении.

```java
public class AsyncJavascriptExample {
    public static void main(String[] args) {
        WebDriver driver = new ChromeDriver();
        JavascriptExecutor js = (JavascriptExecutor) driver;
        driver.get("https://example.com");
        // 1. Простой асинхронный скрипт с задержкой
        js.executeAsyncScript(
            "var callback = arguments[arguments.length - 1]; " +
            "setTimeout(function() { " +
            "  callback('Готово!'); " +
            "}, 2000);"
        );
        System.out.println("Скрипт выполнен после задержки");
        // 2. Асинхронная загрузка данных
        String result = (String) js.executeAsyncScript(
            "var callback = arguments[arguments.length - 1]; " +
            "fetch('https://api.example.com/data') " +
            "  .then(response => response.json()) " +
            "  .then(data => callback(JSON.stringify(data))) " +
            "  .catch(error => callback('Ошибка: ' + error.message));"
        );
        System.out.println("Результат запроса: " + result);
        // 3. Ожидание появления элемента
        js.executeAsyncScript(
            "var elementId = arguments[0]; " +
            "var callback = arguments[arguments.length - 1]; " +
            "function checkElement() { " +
            "  if (document.getElementById(elementId)) { " +
            "    callback('Элемент найден'); " +
            "  } else { " +
            "    setTimeout(checkElement, 100); " +
            "  } " +
            "} " +
            "checkElement();",
            "dynamicElement"
        );
        // 4. Загрузка внешней библиотеки
        js.executeAsyncScript(
            "var callback = arguments[arguments.length - 1]; " +
            "var script = document.createElement('script'); " +
            "script.src = 'https://cdn.jsdelivr.net/npm/lodash@4.17.21/lodash.min.js'; " +
            "script.onload = function() { callback('Библиотека загружена'); }; " +
            "document.head.appendChild(script);"
        );
        System.out.println("Библиотека загружена");
        driver.quit();
    }
}
```

## Продвинутые примеры

### Полезные утилиты на JavaScript

```java
public class AdvancedJavascriptExample {
    private JavascriptExecutor js;
    
    public AdvancedJavascriptExample(WebDriver driver) {
        this.js = (JavascriptExecutor) driver;
    }
    
    // 1. Ожидание загрузки страницы
    public void waitForPageLoad() {
        js.executeScript(
            "var callback = arguments[arguments.length - 1]; " +
            "if (document.readyState === 'complete') { " +
            "  callback(); " +
            "} else { " +
            "  window.addEventListener('load', function() { callback(); }); " +
            "}"
        );
    }
    
    // 2. Ожидание jQuery
    public void waitForJQuery() {
        js.executeAsyncScript(
            "var callback = arguments[arguments.length - 1]; " +
            "if (typeof jQuery === 'undefined') { " +
            "  callback('jQuery не загружен'); " +
            "} else { " +
            "  jQuery.ajaxSetup({ async: false }); " +
            "  callback('jQuery загружен'); " +
            "}"
        );
    }
    
    // 3. Получение позиции элемента
    public Map<String, Long> getElementPosition(WebElement element) {
        Map<String, Long> position = (Map<String, Long>) js.executeScript(
            "var rect = arguments[0].getBoundingClientRect(); " +
            "return { " +
            "  top: rect.top, " +
            "  left: rect.left, " +
            "  width: rect.width, " +
            "  height: rect.height " +
            "}",
            element
        );
        return position;
    }
    
    // 4. Выделение элемента (для отладки)
    public void highlightElement(WebElement element, String color) {
        js.executeScript(
            "var element = arguments[0]; " +
            "var color = arguments[1] || 'red'; " +
            "var originalBorder = element.style.border; " +
            "var originalBackground = element.style.backgroundColor; " +
            "element.style.border = '3px solid ' + color; " +
            "element.style.backgroundColor = color + '33'; " +
            "setTimeout(function() { " +
            "  element.style.border = originalBorder; " +
            "  element.style.backgroundColor = originalBackground; " +
            "}, 2000);",
            element, color
        );
    }
    
    // 5. Скриншот элемента (canvas)
    public String captureElementScreenshot(WebElement element) {
        return (String) js.executeScript(
            "var element = arguments[0]; " +
            "var canvas = document.createElement('canvas'); " +
            "var rect = element.getBoundingClientRect(); " +
            "canvas.width = rect.width; " +
            "canvas.height = rect.height; " +
            "var ctx = canvas.getContext('2d'); " +
            "ctx.drawImage(element, 0, 0); " +
            "return canvas.toDataURL();",
            element
        );
    }
    
    // 6. Проверка наличия элемента
    public boolean isElementPresent(String selector) {
        return (Boolean) js.executeScript(
            "return document.querySelector('" + selector + "') !== null"
        );
    }
    
    // 7. Получение всех данных из таблицы
    public List<List<String>> getTableData(String tableSelector) {
        return (List<List<String>>) js.executeScript(
            "var table = document.querySelector('" + tableSelector + "'); " +
            "if (!table) return []; " +
            "var data = []; " +
            "var rows = table.querySelectorAll('tr'); " +
            "rows.forEach(function(row) { " +
            "  var rowData = []; " +
            "  row.querySelectorAll('td, th').forEach(function(cell) { " +
            "    rowData.push(cell.textContent.trim()); " +
            "  }); " +
            "  data.push(rowData); " +
            "}); " +
            "return data;"
        );
    }
    
    // 8. Работа с localStorage
    public void setLocalStorage(String key, String value) {
        js.executeScript(
            "localStorage.setItem(arguments[0], arguments[1])",
            key, value
        );
    }
    public String getLocalStorage(String key) {
        return (String) js.executeScript(
            "return localStorage.getItem(arguments[0])",
            key
        );
    }
    
    // 9. Работа с sessionStorage
    public void setSessionStorage(String key, String value) {
        js.executeScript(
            "sessionStorage.setItem(arguments[0], arguments[1])",
            key, value
        );
    }
    public String getSessionStorage(String key) {
        return (String) js.executeScript(
            "return sessionStorage.getItem(arguments[0])",
            key
        );
    }
}
```
